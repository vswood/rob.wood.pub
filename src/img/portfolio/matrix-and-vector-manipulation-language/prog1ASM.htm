main ENDP

fill PROC
    pusha
    Call Crlf
readLoop:
    mov dx,offset prompt
    Call Writestring
    Call Readint
    Call Crlf
    mov [bp],ax
    add bp,2
    loop readLoop
    Call Crlf
    popa
    ret
fill ENDP

zFill PROC
    pusha
    mov ax,0
zreadLoop:
    mov [bp],ax
    add bp,2
    loop zreadLoop
    popa
    ret
zFill ENDP

print PROC
    pusha
    Call Crlf
    mov ax,'|'
    Call Writechar
printLoop:
    push dx
    mov dx,[bp]
    mov ax,' '
    cmp dx,0
    jl negative
    Call Writechar
    cmp dx,10
    jge over10
    Call Writechar
over10:
    cmp dx,100
    jge over100
    Call Writechar
over100:
    cmp dx,1000
    jge writeit
    Call Writechar
    jmp writeit
negative:
    cmp dx,-10
    jle overn10
    Call Writechar
overn10:
    cmp dx,-100
    jle overn100
    Call Writechar
overn100:
    cmp dx,-1000
    jle writeit
    Call Writechar
writeit:
    pop dx
    mov ax,[bp]
    mov bx,10
    Call Writeint_signed
    mov ax,'|'
    Call Writechar
    mov ax,cx
    dec ax
    div dl
    cmp ah,0
    jne inline
    cmp cx,1
    je inline
    Call Crlf
    mov ax,'|'
    Call Writechar
inline:
    add bp,2
    loop printLoop
    Call Crlf
    popa
    ret
print ENDP

addArray PROC
    pusha
addLoop:
    mov ax,[bp]
    add ax,ss:[si]
    mov ss:[di],ax
    add bp,2
    add si,2
    add di,2
    loop addLoop
    popa
    ret
addArray ENDP

subArray PROC
    pusha
subLoop:
    mov ax,[bp]
    sub ax,ss:[si]
    mov ss:[di],ax
    add bp,2
    add si,2
    add di,2
    loop subLoop
    popa
    ret
subArray ENDP

scalarMult PROC
    pusha
mulLoop:
    mov ax,[bp]
    mov bx,ss:[si]
    imul bx
    mov ss:[di],ax
    add bp,2
    add di,2
    loop mulLoop
    popa
    ret
scalarMult ENDP

matrixMult PROC
    pusha
    mov ax,0
    mov bx,0
    mov dx,si
multLoop:
    push cx
    mov cx,common
    call calcEntry
    pop cx
    add di,2
    add bp,2
    inc bx
    cmp bx,cols
    je nextRow
    loop multLoop
nextRow:
    inc ax
    cmp ax,rows
    je endLoop
    mov bx,0
    loop multLoop
endLoop:
    popa
    ret
matrixMult ENDP

calcEntry PROC
    pusha
    mov temp,0
calcLoop:
    mov ax,[bp]
    mov bx,ss:[si]
    imul bx
    mov dx,temp
    add dx,ax
    mov temp,dx
    add bp,2
    add si,rowInc
    loop calcLoop
    mov dx,temp
    mov ss:[di],dx
    popa
    ret
calcEntry ENDP

END main
