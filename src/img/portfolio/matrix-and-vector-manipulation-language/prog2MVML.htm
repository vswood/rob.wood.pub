Matrix and Vector Manipulation Language

Sample Program 2 MVML Source Code

mvml

matrix A= matrix(3;2);
matrix B= matrix(2;3);
matrix C= matrix(3;2);
matrix D= matrix(3;3);
vector e= vector(3);
vector f= vector(3);
vector g= vector(3);
vector h= vector(2);
scalar sclr= 2;

out("Fill Matrix A:");
fill(A);
out("Fill Matrix B:");
fill(B);
out("Matrix A=");
print(A);
out("Matrix B=");
print(B);
C=A*sclr;
out("Matrix C=A*sclr");
print(C);
D=A*B;
out("Matrix D=A*B");
print(D);

out("Fill Vector e:");
fill(e);
out("Fill Vector f:");
fill(f);
out("Vector e=");
print(e);
out("Vector f=");
print(f);
g=e*sclr;
out("Vector g=e*sclr");
print(g);
h=B*e;
out("Vector h=B*e");
print(h);

/mvml
